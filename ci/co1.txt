(defun block-2 (x)
   (return-from block-2 x)
)

(if-compiles (compile 'block-2)
   (progn (assert-true "block-2-1"   (eql  100     (block-2  100))  )
   )
)


(defun block-1 (x)
   (block outer
      (+ 3
         (block inner 
            (if (< x 0)
                (return-from outer "minus")
                (if (= x 0)
                    (return-from inner 100)
                    (return-from block-1  9999) ))))))


(if-compiles (compile 'block-1)
   (progn (assert-true "block-1-1"   (string=  "minus"  (block-1  -1))  )
          (assert-true "block-1-2"   (eql      103      (block-1  0 ))  )
          (assert-true "block-1-3"   (eql      9999     (block-1  10))  )
   )
)

;;;   ---- test unwind-protect ----

(defparameter *unw-1* 0)
(defparameter *unw-2* 0)

(defun unw-1 ()
   (unwind-protect
       123

   (assert-true "unw-1-3"   (equal  200    *unw-2* )  )
)

;;;   ---- test java call（段階1: リフレクションで呼ぶ） ----

(java-import '[ java.lang  Integer ])
(java-import '[ java.util  ArrayList ])

(defun jcall-1 (s)
   (.toUpperCase s))

(if-compiles (compile 'jcall-1)
   (assert-true "jcall-1"     (string= "ABC" (jcall-1 "abc")))
)

;; .m は戻り値の boolean を t / nil にする
(defun jcall-2 (s x)
   (.equals s x))

(if-compiles (compile 'jcall-2)
   (assert-true "jcall-2-1"   (eq t   (jcall-2 "abc" "abc")))
   (assert-true "jcall-2-2"   (eq nil (jcall-2 "abc" "xyz")))
)

;; .-m は戻り値を変換しない（Java の Boolean のまま）
(defun jcall-3 (s x)
   (.-equals s x))

(if-compiles (compile 'jcall-3)
   (assert-true "jcall-3-1"   (not (eq t (jcall-3 "abc" "abc"))))
   (assert-true "jcall-3-2"   (eq t (.booleanValue (jcall-3 "abc" "abc"))))
)

;; static メソッド
(defun jcall-4 (s)
   (Integer:parseInt s))

(if-compiles (compile 'jcall-4)
   (assert-true "jcall-4"     (eql 123 (jcall-4 "123")))
)

;; .m の引数: nil → false、nil 以外のシンボル → true、キーワードはそのまま
;; （本体の途中にある式 (.add ...) が文になることのテストも兼ねる）
(defun jcall-5 (lst v)
   (.add lst v)
   (.-get lst 0))

(if-compiles (compile 'jcall-5)
   (assert-true "jcall-5-1"   (eq nil (.booleanValue (jcall-5 (new ArrayList) nil))))
   (assert-true "jcall-5-2"   (eq t   (.booleanValue (jcall-5 (new ArrayList) t))))
   (assert-true "jcall-5-3"   (eq :kw (jcall-5 (new ArrayList) :kw)))
)

;; .-m の引数はシンボルのまま渡す
(defun jcall-6 (lst v)
   (.-add lst v)
   (.-get lst 0))

(if-compiles (compile 'jcall-6)
   (assert-true "jcall-6"     (eq nil (jcall-6 (new ArrayList) nil)))
)

;;;   ---- test java call（段階2: static と new は、候補が1つなら直接呼ぶ） ----

(java-import '[ java.util  HashMap ])

;; コンストラクタ: 引数0個は1つだけ → 直接 new
(defun jcall-7 ()
   (new ArrayList))

(if-compiles (compile 'jcall-7)
   (assert-true "jcall-7"     (eql 0 (.size (jcall-7))))
)

;; コンストラクタ: 引数1個は (int) と (Collection) の2つ → リフレクション
(defun jcall-8 (n)
   (new ArrayList n))

(if-compiles (compile 'jcall-8)
   (assert-true "jcall-8"     (eql 0 (.size (jcall-8 10))))
)

;; 直接呼んだ static メソッドが例外を投げたら、インタプリタと同じくエラー（Condition）になる
(defun jcall-4b (s)
   (Integer:parseInt s))

(if-compiles (compile 'jcall-4b)
   (assert-true "jcall-4b"    (eq :err (handler-case (jcall-4b "abc") (error () :err))))
)

;; static: Math:abs は4つある → リフレクション
(defun jcall-9 (x)
   (Math:abs x))

(if-compiles (compile 'jcall-9)
   (assert-true "jcall-9"     (eql 3 (jcall-9 -3)))
)

;;;   ---- test %null（Java の null） ----

;; 引数の %null は nil に変えずに Java へ渡す
(defun jcall-10 (lst)
   (.add lst %null)
   (.-get lst 0))

(if-compiles (compile 'jcall-10)
   (assert-true "jcall-10"    (let ((v (jcall-10 (new ArrayList))))
                                 (and (null v) (not (eq v nil)))))
)

;; Java が返した null も nil に変えない
(defun jcall-11 (m)
   (list (.get m "no-key")))

(if-compiles (compile 'jcall-11)
   (assert-true "jcall-11"    (let ((v (car (jcall-11 (new HashMap)))))
                                 (and (null v) (not (eq v nil)))))
)

;;;   ---- test java call（段階3: (declare (java-type ...)) で型が分かるインスタンスメソッドは直接呼ぶ） ----

(java-import '[ java.lang  StringBuilder ])

(defun jcall-12 (s)
   (declare (java-type String s))
   (.length s))

(if-compiles (compile 'jcall-12)
   (assert-true "jcall-12"    (eql 3 (jcall-12 "abc")))
)

;; .trim の結果は String と分かるので、.length も直接呼ぶ
(defun jcall-13 (s)
   (declare (java-type String s))
   (.length (.trim s)))

(if-compiles (compile 'jcall-13)
   (assert-true "jcall-13"    (eql 2 (jcall-13 "  ab  ")))
)

;; .m の戻り値の boolean は t / nil になる
(defun jcall-14 (s x)
   (declare (java-type String s))
   (.startsWith s x))

(if-compiles (compile 'jcall-14)
   (assert-true "jcall-14-1"  (eq t   (jcall-14 "abc" "a")))
   (assert-true "jcall-14-2"  (eq nil (jcall-14 "abc" "z")))
)

;; let の宣言
(defun jcall-15 (x)
   (let ((s x))
      (declare (java-type String s))
      (.length s)))

(if-compiles (compile 'jcall-15)
   (assert-true "jcall-15"    (eql 5 (jcall-15 "hello")))
)

;; 宣言と違う型の値を渡したら、インタプリタと同じくエラー
(defun jcall-16 (s)
   (declare (java-type String s))
   (.length s))

(if-compiles (compile 'jcall-16)
   (assert-true "jcall-16"    (eq :err (handler-case (jcall-16 123) (error () :err))))
)

;; char を受け取るメソッド: 文字（LispCharacter）を char に変換して渡す
(defun jcall-17 (sb)
   (declare (java-type StringBuilder sb))
   (.setCharAt sb 0 #\X)
   (.toString sb))

(if-compiles (compile 'jcall-17)
   (assert-true "jcall-17"    (string= "Xbc" (jcall-17 (new StringBuilder "abc"))))
)
