(defun teq (a b)
  (cond
   ((and (stringp a) (stringp b)) (string= a b))
;   ((and (atom a) (atom b)) (eq a b))
   ((and (atom a) (atom b)) (eql a b))
   ((null a) nil)
   ((null b) nil)
   (t (and 
       (teq (car a) (car b)) 
       (teq (cdr a) (cdr b))))))
